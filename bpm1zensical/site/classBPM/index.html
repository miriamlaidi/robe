class BPM:

    def __init__(self,lam_0,n_0,width_x,N_x):

        """
        Creates the simulation grid along x.

        Args:
            lam_0 (Float): wavelength in um 
            n_0 (Float): refractive index of the medium
            width_x (Float): width of the propagation window in um
            N_x (Int): number of points in the x direction
        """
        self.p={}

        self.p['lambda_zero'] = lam_0
        self.p['n_0']=n_0
        self.p['width_propagation']= width_x
        self.p['N_x']= N_x



        #Input Formulas
        self.k_0 = 2*np.pi/(self.p['lambda_zero'])
        self.k= self.k_0*self.p['n_0']


        #Generate the x window



        self.dx = width_x / N_x                       # gap between two points (µm)
        self.x  = (np.arange(N_x) - N_x/2) * self.dx  # centered at zero / 
        self.kx = 2 * np.pi * np.fft.fftfreq(N_x, d=self.dx)



    def gaussian(self,w_0,x_0=0):
        """
        Gaussian input beam E(x, z=0) = exp(-(x - x_0)² / w_0²).

        Args:
            w_0 (float): Waist of the beam, 1/e² radius of the intensity (µm).
            x_0 (float, optional): Position of the beam center (µm). Default is 0.

        Returns:
            ndarray: Complex field on the grid self.x.
        """
        A = np.exp(-(self.x-x_0)**2/(w_0)**2).astype(complex)

        return A


    def propagate(self,A_0,L,dz,delta_n=None,save_every=1):

        """
        Propagation of the field A_0 over a length L using the split-step Fourier method.

        At each step dz, the diffraction operator is applied in Fourier space,
        then the guiding operator (phase due to Δn) is applied in real space.

        Args:
            A_0 (ndarray): Input field E(x, z=0) on the grid self.x.
            L (float): Propagation length (µm).
            dz (float): Propagation step (µm). Requires k_0 * dn_max * dz << 1.
            delta_n (ndarray or callable, optional): Refractive index difference Δn(x).
                Either a fixed array (straight guides), or a function of z returning
                Δn(x) at each step (S-bends). Default is None (free space).
            save_every (int, optional): The intensity is stored every save_every steps
                (1 = every step). Larger values save memory for long structures. Default is 1.

        Returns:
            ndarray: Output field E(x, z=L). The intensity map is stored
                in self.I and the z positions in self.z.
        """

        n_z = int(round(L / dz))    # number of step 

        D_prop = np.exp(-1j*self.kx**2*dz/(2*self.k))          # term of propagation
        if delta_n is None:
            delta_n = np.zeros_like(self.x)                    # free space
        if not callable(delta_n):
            N_index = np.exp(1j * self.k_0 * delta_n * dz)     # fixed index profile: computed once
        A = A_0.copy()
        # intensity map allocated once (much faster than appending one array per step)
        I = np.empty((n_z // save_every + 1, len(self.x)))
        I[0] = np.abs(A)**2


        for j in range(n_z):
            A_hat = np.fft.fft(A)            # 1. pass into the Fourier space
            A_hat = D_prop * A_hat           # 2. apply the diffraction operator in the Fourier space
            A = np.fft.ifft(A_hat)           # 3. return to the real space
            if (j + 1) % save_every == 0:    # store the intensity every save_every steps
                I[(j + 1) // save_every] = np.abs(A)**2

            if callable(delta_n):            # z-dependent profile (S-bends): evaluate Δn(x) at this step
                N_index = np.exp(1j * self.k_0 * delta_n((j + 0.5) * dz) * dz)
            A=N_index*A


        self.z = np.arange(len(I)) * save_every * dz   # one position per stored line of I
        self.I = I

        return A


    def guide(self, W, dn_max, p, x0=0):
        """
        Super-Gaussian index profile Δn(x) of a single guide.

        All the guided structures (straight coupler, S-bends) are built from this profile.

        The profile is only computed for |x - x0| < r_cut, where it is larger than
        dn_max * exp(-50); outside it is set to 0. This makes the function much faster
        (it is called at each step for the S-bends) without changing the result.

        Args:
            W (float): Width of the guide (µm).
            dn_max (float): Maximum refractive index difference.
            p (int): Order of the super-Gaussian (p = 1: Gaussian, large p: close to a step index).
            x0 (float, optional): Position of the guide center (µm). Default is 0.

        Returns:
            ndarray: Index profile Δn(x) on the grid self.x.
        """
        dn = np.zeros_like(self.x)
        r_cut = W / 2 * 50 ** (1 / (2 * p))             # (2 r_cut / W)**(2p) = 50
        i0, i1 = np.searchsorted(self.x, [x0 - r_cut, x0 + r_cut])
        dn[i0:i1] = dn_max * np.exp(-(2 * np.abs(self.x[i0:i1] - x0) / W)**(2 * p))

        return dn


    def fibre_jump_index(self, n_core, n_cladding, a, x0=0):

        """
        Step-index profile of a fiber (absolute index n, not Δn).

        Args:
            n_core (Float): refractive index of the core
            n_cladding (Float): refractive index of the cladding
            a (Float): radius of the core
            x0 (Float): position of the center of the core along x

        Returns:
            ndarray: Refractive index profile n(x) of the fiber.


        """
        n = np.where(np.abs(self.x - x0) <= a, n_core, n_cladding)
        self.n = n
        return n 

    def straight_fiber_coupling(self,W,dn_max,p,gap):

        """
        Index profile Δn(x) of a straight directional coupler: two parallel super-Gaussian
        guides, centered at x1 = -d/2 and x2 = +d/2 with d = W + gap.

        Args:
            W (float): Width of each guide (µm).
            dn_max (float): Maximum refractive index difference.
            p (int): Order of the super-Gaussian; the term (2x/W)**(2p) sets the width
                and the steepness of the edges of each guide.
            gap (float): Edge-to-edge distance between the guides (µm).

        Returns:
            tuple: (dn, x1, x2) with dn the index profile Δn(x) on the grid self.x,
                and x1, x2 the centers of the two guides (µm).
        """
        d = W + gap   # center-to-center distance between the guides
        x1 = -d/2     # center of guide 1
        x2 = d/2      # center of guide 2

        dn = self.guide(W, dn_max, p, x1) + self.guide(W, dn_max, p, x2)

        return dn,x1,x2


    def s_bend_center(self, z, R, h, L0=0, z_start=0):

        """
        Center position of a guide with an S-bend in, a straight section, and an S-bend out.

        Args:
            z (float): Position along the propagation axis (µm).
            R (float): Radius of curvature of the arcs (µm).
            h (float): Lateral offset of each S-bend (µm).
            L0 (float, optional): Length of the straight section between the two S-bends (µm). Default is 0.
            z_start (float, optional): Position where the S-bend in starts (µm). Default is 0.

        Returns:
            float: Position x_c of the guide center (µm).

        """
        Ls = np.sqrt(4 * R * h - h**2)        # length of one S-bend
        zz = z - z_start                      # z measured from the start of the S-bend in
        u = zz - Ls - L0                      # z measured from the start of the S-bend out

        #  before 
        if zz <= 0:
            return 0.0

        #  S-bend in: from 0 to h 
        elif zz <= Ls / 2:                    # first arc
            return R - np.sqrt(R**2 - zz**2)
        elif zz <= Ls:                        # second arc
            return h - (R - np.sqrt(R**2 - (Ls - zz)**2))

        # straight section 
        elif zz <= Ls + L0:
            return h

        #  S-bend out: from h back to 0 
        elif u <= Ls / 2:                     # first arc
            return h - (R - np.sqrt(R**2 - u**2))
        elif u <= Ls:                         # second arc
            return R - np.sqrt(R**2 - (Ls - u)**2)

        # after 
        else:
            return 0.0

    def s_bend_profile(self, z, W, dn_max, p, R, h, L0=0, z_start=0):
        """Refractive index profile Δn(x) of an S-bend guide at position z.

        Args:
            z (float): Position along the propagation axis (µm).
            W (float): Width of the guide (µm).
            dn_max (float): Maximum refractive index difference.
            p (int): Order of the super-Gaussian.
            R (float): Radius of curvature (µm).
            h (float): Lateral offset of each S-bend (µm).
            L0 (float, optional): Length of the straight section between the two S-bends (µm). Default is 0.
            z_start (float, optional): Position where the S-bend in starts (µm). Default is 0.

        Returns:
            ndarray: Index profile Δn(x) on the grid self.x.
        """
        xc = self.s_bend_center(z, R, h, L0=L0, z_start=z_start)   # guide position at this z
        return self.guide(W, dn_max, p, xc)


    def s_bend_coupler_profile(self, z, W, dn_max, p, gap, R, h, L0=0, z_start=0):

        """
        Refractive index profile Δn(x) of a directional coupler with two opposite S-bends at position z.

        Args:
            z (float): Position along the propagation axis (µm).
            W (float): Width of each guide (µm).
            dn_max (float): Maximum refractive index difference.
            p (int): Order of the super-Gaussian.
            gap (float): Edge-to-edge distance between the guides in the straight section (µm).
            R (float): Radius of curvature of the S-bends (µm).
            h (float): Lateral offset of each S-bend (µm).
            L0 (float, optional): Length of the straight coupling section (µm). Default is 0.
            z_start (float, optional): Position where the S-bends in start (µm). Default is 0.

        Returns:
            tuple: (dn, x1, x2) with dn the index profile Δn(x) on the grid self.x,
                and x1, x2 the centers of the two guides at this z (µm).

        """
        d = W + gap                                                # center-to-center distance in the straight section
        s = self.s_bend_center(z, R, h, L0=L0, z_start=z_start)    # offset of the S-bend at this z (0 → h → 0)

        x1 = -(d / 2 + h) + s                                      # guide 1 goes up
        x2 = +(d / 2 + h) - s                                      # guide 2 goes down (opposite S-bend)

        dn = self.guide(W, dn_max, p, x1) + self.guide(W, dn_max, p, x2)

        return dn, x1, x2


    def mach_zehnder_profile(self, z, W, dn_max, p, gap, R, h, L_c, L_arm, dn_phase, z_start=0):
        """
        Refractive index profile Δn(x) of a Mach-Zehnder interferometer at position z:
        two S-bent couplers separated by two arms, with a phase shifter on the arm of guide 1.

        Args:
            z (float): Position along the propagation axis (µm).
            W (float): Width of each guide (µm).
            dn_max (float): Maximum refractive index difference of the guides.
            p (int): Order of the super-Gaussian.
            gap (float): Edge-to-edge distance between the guides inside the couplers (µm).
            R (float): Radius of curvature of the S-bends (µm).
            h (float): Lateral offset of each S-bend (µm).
            L_c (float): Length of the straight section of each coupler (µm).
            L_arm (float): Length of the arms between the two couplers (µm).
            dn_phase (float): Extra index added on the arm of guide 1 (phase shifter).
            z_start (float, optional): Position where the first coupler starts (µm). Default is 0.

        Returns:
            tuple: (dn, x1, x2) with dn the index profile Δn(x) on the grid self.x,
                and x1, x2 the centers of the two guides at this z (µm).
        """
        d = W + gap
        Ls = np.sqrt(4*R*h - h**2)

        z1 = z_start          # start of the first s-bend
        z2 = z1 + 2*Ls + L_c  # end of the  fisrt s coupler
        z3 = z2 + L_arm       # end of the mach-zehnder arm/ sart of the second s-bend
        z4 = z3 + 2*Ls + L_c  # end of the second s coupler


        # lateral offset: first coupler (starts at z1) + second coupler (starts at z3)
        # s_bend_center returns 0 outside its coupler, so the two terms can be added
        s = (self.s_bend_center(z, R, h, L0=L_c, z_start=z1)
             + self.s_bend_center(z, R, h, L0=L_c, z_start=z3))

        x1 = -(d / 2 + h) + s     # guide 1 (bottom) goes up in the couplers
        x2 = +(d / 2 + h) - s     # guide 2 (top) goes down in the couplers

        dn = self.guide(W, dn_max, p, x1) + self.guide(W, dn_max, p, x2)

        # phase shifter: extra index on guide 1, only in the arm (between z2 and z3)
        if z2 <= z < z3:
            dn = dn + self.guide(W, dn_phase, p, x1)

        return dn, x1, x2

        W (float): Width of the guide (µm).
        dn_max (float): Maximum refractive index difference.
        p (int): Order of the super-Gaussian (p = 1: Gaussian, large p: close to a step index).
        x0 (float, optional): Position of the guide center (µm). Default is 0.

    Returns:
        ndarray: Index profile Δn(x) on the grid self.x.
    """
    dn = np.zeros_like(self.x)
    r_cut = W / 2 * 50 ** (1 / (2 * p))             # (2 r_cut / W)**(2p) = 50
    i0, i1 = np.searchsorted(self.x, [x0 - r_cut, x0 + r_cut])
    dn[i0:i1] = dn_max * np.exp(-(2 * np.abs(self.x[i0:i1] - x0) / W)**(2 * p))

    return dn

def s_bend_center(self, z, R, h, L0=0, z_start=0):

    """
    Center position of a guide with an S-bend in, a straight section, and an S-bend out.

    Args:
        z (float): Position along the propagation axis (µm).
        R (float): Radius of curvature of the arcs (µm).
        h (float): Lateral offset of each S-bend (µm).
        L0 (float, optional): Length of the straight section between the two S-bends (µm). Default is 0.
        z_start (float, optional): Position where the S-bend in starts (µm). Default is 0.

    Returns:
        float: Position x_c of the guide center (µm).

    """
    Ls = np.sqrt(4 * R * h - h**2)        # length of one S-bend
    zz = z - z_start                      # z measured from the start of the S-bend in
    u = zz - Ls - L0                      # z measured from the start of the S-bend out

    #  before 
    if zz <= 0:
        return 0.0

    #  S-bend in: from 0 to h 
    elif zz <= Ls / 2:                    # first arc
        return R - np.sqrt(R**2 - zz**2)
    elif zz <= Ls:                        # second arc
        return h - (R - np.sqrt(R**2 - (Ls - zz)**2))

    # straight section 
    elif zz <= Ls + L0:
        return h

    #  S-bend out: from h back to 0 
    elif u <= Ls / 2:                     # first arc
        return h - (R - np.sqrt(R**2 - u**2))
    elif u <= Ls:                         # second arc
        return R - np.sqrt(R**2 - (Ls - u)**2)

    # after 
    else:
        return 0.0

def s_bend_coupler_profile(self, z, W, dn_max, p, gap, R, h, L0=0, z_start=0):

    """
    Refractive index profile Δn(x) of a directional coupler with two opposite S-bends at position z.

    Args:
        z (float): Position along the propagation axis (µm).
        W (float): Width of each guide (µm).
        dn_max (float): Maximum refractive index difference.
        p (int): Order of the super-Gaussian.
        gap (float): Edge-to-edge distance between the guides in the straight section (µm).
        R (float): Radius of curvature of the S-bends (µm).
        h (float): Lateral offset of each S-bend (µm).
        L0 (float, optional): Length of the straight coupling section (µm). Default is 0.
        z_start (float, optional): Position where the S-bends in start (µm). Default is 0.

    Returns:
        tuple: (dn, x1, x2) with dn the index profile Δn(x) on the grid self.x,
            and x1, x2 the centers of the two guides at this z (µm).

    """
    d = W + gap                                                # center-to-center distance in the straight section
    s = self.s_bend_center(z, R, h, L0=L0, z_start=z_start)    # offset of the S-bend at this z (0 → h → 0)

    x1 = -(d / 2 + h) + s                                      # guide 1 goes up
    x2 = +(d / 2 + h) - s                                      # guide 2 goes down (opposite S-bend)

    dn = self.guide(W, dn_max, p, x1) + self.guide(W, dn_max, p, x2)

    return dn, x1, x2

def mach_zehnder_profile(self, z, W, dn_max, p, gap, R, h, L_c, L_arm, dn_phase, z_start=0):
    """
    Refractive index profile Δn(x) of a Mach-Zehnder interferometer at position z:
    two S-bent couplers separated by two arms, with a phase shifter on the arm of guide 1.

    Args:
        z (float): Position along the propagation axis (µm).
        W (float): Width of each guide (µm).
        dn_max (float): Maximum refractive index difference of the guides.
        p (int): Order of the super-Gaussian.
        gap (float): Edge-to-edge distance between the guides inside the couplers (µm).
        R (float): Radius of curvature of the S-bends (µm).
        h (float): Lateral offset of each S-bend (µm).
        L_c (float): Length of the straight section of each coupler (µm).
        L_arm (float): Length of the arms between the two couplers (µm).
        dn_phase (float): Extra index added on the arm of guide 1 (phase shifter).
        z_start (float, optional): Position where the first coupler starts (µm). Default is 0.

    Returns:
        tuple: (dn, x1, x2) with dn the index profile Δn(x) on the grid self.x,
            and x1, x2 the centers of the two guides at this z (µm).
    """
    d = W + gap
    Ls = np.sqrt(4*R*h - h**2)

    z1 = z_start          # start of the first s-bend
    z2 = z1 + 2*Ls + L_c  # end of the  fisrt s coupler
    z3 = z2 + L_arm       # end of the mach-zehnder arm/ sart of the second s-bend
    z4 = z3 + 2*Ls + L_c  # end of the second s coupler


    # lateral offset: first coupler (starts at z1) + second coupler (starts at z3)
    # s_bend_center returns 0 outside its coupler, so the two terms can be added
    s = (self.s_bend_center(z, R, h, L0=L_c, z_start=z1)
         + self.s_bend_center(z, R, h, L0=L_c, z_start=z3))

    x1 = -(d / 2 + h) + s     # guide 1 (bottom) goes up in the couplers
    x2 = +(d / 2 + h) - s     # guide 2 (top) goes down in the couplers

    dn = self.guide(W, dn_max, p, x1) + self.guide(W, dn_max, p, x2)

    # phase shifter: extra index on guide 1, only in the arm (between z2 and z3)
    if z2 <= z < z3:
        dn = dn + self.guide(W, dn_phase, p, x1)

    return dn, x1, x2
