Class BPMplot¶

Graphical interface (NiceGUI) for the BPM simulations¶

This class builds an input panel (sliders for the beam and the guide parameters, medium selector, injection options) and an output panel with the intensity map \(|A(x,z)|^2\) and a status line. The calculation is delegated to the functions of each case (free space, single waveguide, straight coupler, S-bent coupler, Mach-Zehnder), which use the BPM class.

def draw(r, title='', show_title=True):
    """
        Plot intensity map in the current figure. The plot has z on the horizontal axis and x on the vertical axis, the vertical
        range is limited to ±x_lim to show only the useful region.

        Args:
            r(Dict): Dictionnary with all the arrays and parameters we need to draw the figure (intensity map I, axes z and x,
                  guide edges, x_lim, info text).
            title(String): Title of the figure.
        Returns:
            Figure.
    """
    plt.clf() #clear figure
    plt.imshow(r['I'].T,
                   extent=[r['z'][0], r['z'][-1], r['x'][0], r['x'][-1]], #edges
                   origin='lower', aspect='auto', cmap='inferno') #origin
    for zc, xc in r['guides']:                  
            plt.plot(zc, xc, color='w', lw=0.6, ls='--') #dotted lines for the guides
    plt.xlim(r['z'][0], r['z'][-1])
    plt.ylim(-r['x_lim'], r['x_lim']) 
    plt.xlabel('z (µm)')
    plt.ylabel('x (µm)')
    if show_title:
        plt.title(title)
    plt.colorbar(label='Intensity |A|²')
    plt.tight_layout()

class BPMplot:
    """
            Class for GUI interface.

    """
    def __init__(self, show_title=True): #method that runs automatically when an object is created and sets up its initial state
        self.show_title = show_title #showing title only for the first plot
        self.build_ui()
        ui.timer(0.5, self.update, once=True)

    def slider(self, text, unit, fmt, **kw):
        """
                Builds the sliders and the corresponding label.

                Args:
                    self(Object) 
                    text(Str): Name displayed in the label (e.g. 'w0', 'Gap').
                    unit(Str): Unit appended to the value (e.g. 'µm'); use '' if there is none.
                    fmt(Str): Format of the displayed number (e.g. '.1f' for one decimal place).
                    **kw(Dict): Options passed to ui.slider (min, max, step, value).

                Returns:
                    tuple: (slider, column)
            """
        with ui.column().classes('w-full gap-0') as col:
                    lbl = ui.label()
                    s = ui.slider(**kw).props('label')
                    lbl.bind_text_from(s, 'value', lambda v: f'{text} = {v:{fmt}} {unit}'.strip())
        return s, col

    def build_ui(self):
        """
                Calculates and draws using the different functions for each case.

                Args:
                    Self(Object):
                Returns:
                    Figure.
            """
        if self.show_title:
            ui.label('Laser beam propagation — BPM').classes('text-2xl font-bold') #only for the first plot

        with ui.row().classes('w-full no-wrap items-start'):

            ##input
            with ui.card().classes('w-80 shrink-0'):
                ui.label('Beam').classes('text-lg font-semibold')
                self.w0, _ = self.slider('w0', 'µm', '.1f', min=1, max=30, step=0.5, value=3)
                self.lam, _ = self.slider('λ', 'µm', '.2f', min=0.4, max=1.6, step=0.01, value=1.55)
                self.x0, _ = self.slider('Beam offset x0', 'µm', '.1f', min=-10, max=10, step=0.5, value=0)
                self.z, _ = self.slider('Propagation distance z', 'µm', '.0f', min=100, max=20000, step=50, value=500)

                ui.separator()
                ui.label('Medium').classes('text-lg font-semibold')
                self.medium = ui.select(list(CASES.keys()), value='Free space').classes('w-full')

                #sliders
                self.W, col_W = self.slider('Guide width W', 'µm', '.1f', min=1, max=20, step=0.5, value=4)
                self.dn_max, col_dn = self.slider('Δn max', '', '.3f', min=0.001, max=0.02, step=0.001, value=0.01)
                self.gap, col_gap = self.slider('Gap', 'µm', '.1f', min=1, max=15, step=0.5, value=4)
                self.L, col_L = self.slider('Interaction length L', 'µm', '.0f', min=0, max=3000, step=50, value=700)
                self.h, col_h = self.slider('S-bend offset h', 'µm', '.0f', min=2, max=40, step=1, value=H)
                self.L_arm, col_arm = self.slider('Arm length L_arm', 'µm', '.0f', min=0, max=3000, step=50, value=1000)
                self.dn_phase, col_ph = self.slider('Phase shifter Δn', '', '.5f', min=0, max=0.002, step=0.00005, value=0)

                self.n_guides = ui.toggle({1: '1 guide', 2: '2 guides'}, value=2)
                self.injection = ui.toggle({'guide 1': 'Bottom', 'guide 2': 'Top', 'both': 'Both'},
                                            value='guide 1')

                #is_coupler = lambda m: m.startswith('Coupler')
                col_W.bind_visibility_from(self.medium, 'value', backward=lambda m: m != 'Free space')
                col_dn.bind_visibility_from(self.medium, 'value', backward=lambda m: m != 'Free space')
                col_gap.bind_visibility_from(self.medium, 'value', backward=is_coupler)
                col_L.bind_visibility_from(self.medium, 'value', backward=is_s_bend)
                col_h.bind_visibility_from(self.medium, 'value', backward=is_s_bend)
                col_arm.bind_visibility_from(self.medium, 'value', backward=is_mz)
                col_ph.bind_visibility_from(self.medium, 'value', backward=is_mz)
                self.n_guides.bind_visibility_from(self.medium, 'value',
                                                backward=lambda m: is_coupler(m) and not is_mz(m))
                self.injection.bind_visibility_from(self.medium, 'value', backward=is_coupler)

                # z suggested when we change medium, L or h
                self.medium.on_value_change(self.suggest_z)
                for s in (self.L, self.h, self.L_arm):
                    s.on_value_change(lambda: is_s_bend(self.medium.value) and self.suggest_z())
                ui.separator()
                with ui.row().classes('items-center'):
                    self.button = ui.button('Simulate', icon='play_arrow', on_click=self.update)
                    self.spinner = ui.spinner(size='lg')
                    self.spinner.visible = False

            ##output
            with ui.card().classes('grow'):
                with ui.pyplot(figsize=(9, 4.5), close=False) as self.fig:
                    pass
                self.status = ui.label('Choose the parameters and press Simulate.') \
                                .classes('text-gray-600')

    def suggest_z(self):
        """
                                Set the propagation distance slider to a suggested value for the selected medium.

                                Returns:
                                    None: the value is written directly in the z slider (self.z.value).
                            """
        m = self.medium.value
        if is_mz(m):
            self.z.value = int(np.ceil(mz_length(self.L.value, self.L_arm.value, self.h.value) / 50) * 50)
        elif is_s_bend(m):
            self.z.value = int(np.ceil(s_bend_length(self.L.value, self.h.value) / 50) * 50)
        else:
            self.z.value = DEFAULT_Z[m]

    def get_params(self):
        """
                        Collect the current values of all the sliders and toggles in a dictionary.

                        Returns:
                            dict: Parameters of the simulation, with the keys lam, w0, z, W, dn_max,
        x0, gap, L, h, L_arm, dn_phase, n_guides and injection.
                    """
        return dict(lam=self.lam.value, w0=self.w0.value, z=self.z.value,
                W=self.W.value, dn_max=self.dn_max.value, x0=self.x0.value,
                gap=self.gap.value, L=self.L.value, h=self.h.value,
                L_arm=self.L_arm.value, dn_phase=self.dn_phase.value,
                n_guides=self.n_guides.value, injection=self.injection.value)

    async def update(self):
            """
                                    Run the simulation of the selected medium and refresh the plot.

    The interface stays responsive during the calculation: the button is disabled
    and a spinner and a "Computing…" message are shown, while the case function runs
    in a background thread (run.io_bound). When it finishes, the intensity map is
    drawn on the figure and the status label shows the output information
    (e.g. power in each guide) and the computation time. Any error is printed in
    the terminal and shown in the status label instead of stopping the app.


                                    Returns:
                                        Updates.
                                """
            self.button.disable()
            self.spinner.visible = True
            self.status.text = 'Computing…'
            t0 = time.perf_counter()
            try:
                r = await run.io_bound(CASES[self.medium.value], self.get_params())
                title = f'{self.medium.value} — w0 = {self.w0.value} µm, λ = {self.lam.value} µm'
                with self.fig:
                    draw(r, title, self.show_title)
                self.status.text = f"{r['info']}   ({time.perf_counter() - t0:.1f} s)"
            except Exception as e:
                traceback.print_exc()
                self.status.text = f'Error: {e}'
            finally:
                self.button.enable()
                self.spinner.visible = False
